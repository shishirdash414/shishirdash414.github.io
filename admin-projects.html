import { onAuthStateChanged } from "https://www.gstatic.com/firebasejs/12.1.0/firebase-auth.js"; import { getFirestore, collection, addDoc, serverTimestamp } from "https://www.gstatic.com/firebasejs/12.1.0/firebase-firestore.js"; import { auth } from "./firebase-config.js"; const db = getFirestore(); const projectForm = document.getElementById("projectForm"); const formMessage = document.getElementById("formMessage"); console.log("admin-projects.js loaded"); onAuthStateChanged(auth, (user) => { console.log("Auth state:", user); if (!user) { console.log("No authenticated user."); window.location.href = "admin.html"; return; } console.log("Logged in as:", user.email); }); projectForm.addEventListener("submit", async (event) => { event.preventDefault(); console.log("Save Project button clicked"); const user = auth.currentUser; if (!user) { console.error("No authenticated user."); formMessage.textContent = "You are not logged in."; formMessage.style.color = "#ff7070"; return; } const title = document.getElementById("projectTitle").value.trim(); const category = document.getElementById("projectCategory").value; const description = document.getElementById("projectDescription").value.trim(); const technologiesInput = document.getElementById("projectTechnologies").value.trim(); const github = document.getElementById("projectGithub").value.trim(); const demo = document.getElementById("projectDemo").value.trim(); const image = document.getElementById("projectImage").value.trim(); const featured = document.getElementById("projectFeatured").checked; const technologies = technologiesInput ? technologiesInput .split(",") .map(item => item.trim()) .filter(item => item !== "") : []; console.log("Project data:", { title, category, description, technologies, github, demo, image, featured }); formMessage.textContent = "Saving project..."; formMessage.style.color = "#a5a8b3"; try { console.log("Writing to Firestore..."); const docRef = await addDoc( collection(db, "projects"), { title, category, description, technologies, github, demo, image, featured, createdBy: user.uid, createdAt: serverTimestamp() } ); console.log( "Project saved successfully:", docRef.id ); formMessage.textContent = "Project saved successfully."; formMessage.style.color = "#8b7cff"; projectForm.reset(); } catch (error) { console.error( "FIRESTORE ERROR:", error ); formMessage.textContent = "Error: " + error.message; formMessage.style.color = "#ff7070"; } });
